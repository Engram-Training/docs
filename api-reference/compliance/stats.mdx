---
title: "Get Stats"
api: "GET /v1/compliance/stats"
description: "Get aggregate compliance statistics. Public endpoint — no authentication required."
---

## Response

<ResponseField name="compliance" type="object">
  <Expandable>
    <ResponseField name="totalAddresses" type="number">Total indexed addresses across all sources.</ResponseField>
    <ResponseField name="bySource" type="object">Address count per data source.</ResponseField>
    <ResponseField name="lastIngestion" type="object">Most recent data sync timestamp and source.</ResponseField>
    <ResponseField name="totalScreenings" type="number">All-time screening requests.</ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="community" type="object">
  <Expandable>
    <ResponseField name="totalReports" type="number">Total community reports submitted.</ResponseField>
    <ResponseField name="byStatus" type="object">Reports by status (pending/confirmed/rejected).</ResponseField>
    <ResponseField name="byCategory" type="object">Reports by category.</ResponseField>
  </Expandable>
</ResponseField>

<RequestExample>
```bash cURL
curl https://api.engram.training/v1/compliance/stats
```
</RequestExample>
